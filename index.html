Modelly: agent-based modeling in Python#

Agent-based models in Python, fast enough to explore.

Modelly is an open-source framework for agent-based modeling. Write agents with a step(), keep their state in numpy columns as the model grows, and get the same result from the same seed every time.

class Person(CellAgent):
    def step(self):
        if self.state != SUSCEPTIBLE:
            return
        near = self.cell.neighborhood.agents
        sick = sum(a.state == INFECTED for a in near)
        risk = 1 - (1 - 0.12) ** sick
        if self.random.random() < risk:
            self.state = INFECTED

    A model in three pieces

    Boltzmann's wealth model: agents wander a grid and hand a coin to whoever shares their cell.

    1. A scenario holds the inputs: population, grid size, seed.
    2. Agents each know one thing: how to take their own step.
    3. A model owns the world and moves it forward in time.
    from modelly import Model
from modelly.space import CellAgent, OrthogonalMooreGrid
from modelly.scenarios import Scenario


class WealthScenario(Scenario):
    n: int = 120
    width: int = 12
    height: int = 12


class MoneyAgent(CellAgent):
    def __init__(self, model, cell):
        super().__init__(model)
        self.cell = cell
        self.wealth = 1

    def step(self):
        self.cell = self.cell.neighborhood.select_random_cell()
        cellmates = [a for a in self.cell.agents if a is not self]
        if self.wealth > 0 and cellmates:
            self.random.choice(cellmates).wealth += 1
            self.wealth -= 1


class WealthModel(Model):
    def __init__(self, scenario: WealthScenario = WealthScenario):
        super().__init__(scenario=scenario)
        self.grid = OrthogonalMooreGrid(
            (scenario.width, scenario.height), random=self.random
        )
        cells = self.random.choices(self.grid.all_cells.cells, k=scenario.n)
        MoneyAgent.create_agents(self, scenario.n, cells)

    def step(self):
        self.agents.shuffle_do("step")


model = WealthModel()
for _ in range(100):
    model.step()
